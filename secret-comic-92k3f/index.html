<body>

  <div class="wrap">

    <header>

      <button id="prev">◀ Prev</button>

      <button id="next">Next ▶</button>

      <span class="counter" id="counter">1 / 1</span>

      <div style="margin-left:auto"></div>

      <button id="fs">⤢ Fullscreen</button>

    </header>


    <main class="viewer" id="viewer">

      <img id="page" class="page" alt="Page">

    </main>


    <footer>

      <div class="thumbs" id="thumbs"></div>

    </footer>

  </div>


  <script>

    // List your images (relative to this file)

    const PAGES = [

      "images/page01.jpg",

      "images/page02.jpg",

      "images/page03.jpg"

    ];


    let i = 0;

    const img = document.getElementById('page');

    const ctr = document.getElementById('counter');

    const thumbs = document.getElementById('thumbs');


    function clamp(x){ return Math.max(0, Math.min(PAGES.length-1, x)); }

    function set(iNew){

      i = clamp(iNew);

      img.src = PAGES[i];

      ctr.textContent = (i+1) + " / " + PAGES.length;

      [...thumbs.children].forEach((t, k)=>t.classList.toggle('active', k===i));

      // preload neighbors

      [i-1,i+1].forEach(j=>{ if(j>=0 && j<PAGES.length){ const p=new Image(); p.src=PAGES[j]; }});

    }


    document.getElementById('prev').onclick = ()=> set(i-1);

    document.getElementById('next').onclick = ()=> set(i+1);

    document.getElementById('fs').onclick = ()=> {

      if (!document.fullscreenElement) document.documentElement.requestFullscreen?.();

      else document.exitFullscreen?.();

    };


    // thumbs

    PAGES.forEach((src, k)=>{

      const t=document.createElement('img');

      t.src=src; t.className='thumb'+(k===0?' active':''); t.alt='Page '+(k+1);

      t.onclick=()=> set(k);

      thumbs.appendChild(t);

    });


    // keys + swipe

    window.addEventListener('keydown', e => {

      if(e.key==='ArrowRight') set(i+1);

      if(e.key==='ArrowLeft')  set(i-1);

    });

    let startX=null;

    document.getElementById('viewer').addEventListener('touchstart',e=>{

      if(e.touches?.length) startX=e.touches[0].clientX;

    }, {passive:true});

    document.getElementById('viewer').addEventListener('touchend',e=>{

      if(startX===null) return;

      const endX=e.changedTouches?.[0]?.clientX ?? startX;

      const dx=endX-startX;

      if(dx<-40) set(i+1);

      if(dx> 40) set(i-1);

      startX=null;

    }, {passive:true});


    set(0);

  </script>
